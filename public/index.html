    <!-- SECTION 2: DASHBOARD (Hidden until logged in) -->

                <!-- Dependent accounts load dynamically -->

            </select>

<script>

    let currentUser = null;

    let activeTargetAccountId = null; // Tracks selected dependent ID for guardians


    // Helper for API requests with JWT and target account header

    async function apiFetch(url, options = {}) {

        const token = localStorage.getItem('token');

        const headers = {

            'Content-Type': 'application/json',

            'Authorization': `Bearer ${token}`,

            ...options.headers

        };


        if (activeTargetAccountId) {

            headers['x-target-account-id'] = activeTargetAccountId;

        }


        const response = await fetch(url, { ...options, headers });

        return response;

    }


    // 1. Handle User Login

    async function handleLogin(event) {

        event.preventDefault();

        const email = document.getElementById('email').value;

        const password = document.getElementById('password').value;


        try {

            const response = await fetch('/api/auth/login', {

                method: 'POST',

                headers: { 'Content-Type': 'application/json' },

                body: JSON.stringify({ email, password })

            });


            const data = await response.json();


            if (response.ok && data.token) {

                localStorage.setItem('token', data.token);

                localStorage.setItem('user', JSON.stringify(data.user));

                initDashboard(data.user);

            } else {

                alert('Login failed: ' + (data.error || 'Check credentials'));

            }

        } catch (err) {

            console.error('Error logging in:', err);

            alert('Server error during login.');

        }

    }


    // 2. Load Managed Dependents for Switcher

    async function loadDependents() {

        try {

            const response = await apiFetch('/api/guardians/dependents');

            if (!response.ok) return;


            const dependents = await response.json();

            const selector = document.getElementById('accountSelector');

            

            // Keep default "Self" option, clear others

            selector.innerHTML = '<option value="">My Account (Self)</option>';


            dependents.forEach(dep => {

                const option = document.createElement('option');

                option.value = dep.id;

                option.textContent = `${dep.email} (${dep.permission_level} access)`;

                selector.appendChild(option);

            });

        } catch (err) {

            console.error('Could not load dependents:', err);

        }

    }


    // 3. Handle Account Switcher Change

    function handleAccountSwitch(targetId) {

        activeTargetAccountId = targetId ? parseInt(targetId) : null;

        loadActivationCode();

    }


    // 4. Save Carrier Preference

    async function saveCarrier() {

        const carrierName = document.getElementById('carrierSelect').value;

        const targetId = activeTargetAccountId || currentUser.id;


        try {

            const response = await apiFetch('/api/user/carrier', {

                method: 'PUT',

                body: JSON.stringify({ userId: targetId, carrierName })

            });


            if (response.ok) {

                alert('Carrier updated successfully!');

                loadActivationCode();

            } else {

                alert('Failed to update carrier.');

            }

        } catch (err) {

            console.error('Error saving carrier:', err);

        }

    }


    // 5. Fetch Forwarding Code for One-Tap Button

    async function loadActivationCode() {

        if (!currentUser) return;

        const targetId = activeTargetAccountId || currentUser.id;


        try {

            const response = await apiFetch(`/api/user/activation-code?userId=${targetId}`);

            const data = await response.json();


            const btn = document.getElementById('activateBtn');

            if (response.ok && data.telUri) {

                btn.href = data.telUri;

                btn.classList.remove('hidden');

            } else {

                btn.classList.add('hidden');

            }

        } catch (err) {

            console.error('Error fetching activation code:', err);

        }

    }


    // 6. Initialize Dashboard View

    function initDashboard(user) {

        currentUser = user;

        document.getElementById('loginSection').classList.add('hidden');

        document.getElementById('dashboardSection').classList.remove('hidden');

        document.getElementById('welcomeMsg').innerText = `Hello, ${user.name || user.email}`;


        if (user.carrierName) {

            document.getElementById('carrierSelect').value = user.carrierName;

        }

        

        loadDependents();

        loadActivationCode();

    }


    // 7. Logout Session

    function handleLogout() {

        localStorage.clear();

        location.reload();

    }


    // Auto-login on load if token exists

    window.onload = () => {

        const savedUser = localStorage.getItem('user');

        const token = localStorage.getItem('token');

        if (savedUser && token) {

            initDashboard(JSON.parse(savedUser));

        }

    };

</script>
